# Build the Image Classification Model

# 1. Import TensorFlow Library

import tensorflow as tf
from tensorflow import keras
import matplotlib.pyplot as plt
import numpy as np

# 2. Load MNIST Dataset

(train_images, train_labels), (test_images, test_labels) = keras.datasets.mnist.load_data()

# 3. Display Dataset Shape

print("Training Images Shape:", train_images.shape)
print("Training Labels Shape:", train_labels.shape)
print("Testing Images Shape:", test_images.shape)
print("Testing Labels Shape:", test_labels.shape)

# 4. Display First Image

plt.imshow(train_images[0], cmap='gray')
plt.title("Label: " + str(train_labels[0]))
plt.axis("off")
plt.show()

# 5. Normalize Images

train_images = train_images / 255.0
test_images = test_images / 255.0

# 6. Reshape Images for CNN

train_images = train_images.reshape(-1, 28, 28, 1)
test_images = test_images.reshape(-1, 28, 28, 1)

# 7. Display New Shape

print("Training Images Shape:", train_images.shape)
print("Testing Images Shape:", test_images.shape)

# 8. Define CNN Model Architecture

model = keras.Sequential([
    keras.layers.Conv2D(
        32,
        (3, 3),
        activation='relu',
        input_shape=(28, 28, 1)
    ),

    keras.layers.MaxPooling2D((2, 2)),

    keras.layers.Flatten(),

    keras.layers.Dense(128, activation='relu'),

    keras.layers.Dense(10, activation='softmax')
])

# 9. Compile the Model

model.compile(
    optimizer='sgd',
    loss='sparse_categorical_crossentropy',
    metrics=['accuracy']
)

# 10. Train the Model

history = model.fit(
    train_images,
    train_labels,
    epochs=5
)

# 11. Evaluate the Model

test_loss, test_accuracy = model.evaluate(
    test_images,
    test_labels
)

print("Test Loss:", test_loss)
print("Test Accuracy:", test_accuracy)

# 12. Predict the Digit

predictions = model.predict(test_images)

predicted_digit = np.argmax(predictions[0])

print("Predicted Digit:", predicted_digit)
print("Actual Digit:", test_labels[0])

# 13. Display Predicted Image

plt.imshow(test_images[0].reshape(28, 28), cmap="gray")

plt.title(
    "Predicted: {} Actual: {}".format(
        predicted_digit,
        test_labels[0]
    )
)

plt.axis("off")
plt.show()

# 14. Plot Model Loss

plt.plot(history.history['loss'])
plt.title("Model Loss")
plt.xlabel("Epoch")
plt.ylabel("Loss")
plt.show()

# 15. Plot Model Accuracy

plt.plot(history.history['accuracy'])
plt.title("Model Accuracy Graph")
plt.ylabel("Accuracy")
plt.xlabel("Epoch")
plt.show()
