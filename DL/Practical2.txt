# Implementing Feedforward Neural Networks with Keras and TensorFlow

# 1. Import necessary packages

import tensorflow as tf
from tensorflow import keras
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import random

%matplotlib inline

# 2. Load the Fashion MNIST dataset

mnist = tf.keras.datasets.fashion_mnist

(x_train, y_train), (x_test, y_test) = mnist.load_data()

# 3. Normalize the data

x_train = x_train / 255.0
x_test = x_test / 255.0

# 4. Define the network architecture

model = keras.Sequential([
    keras.layers.Flatten(input_shape=(28, 28)),
    keras.layers.Dense(128, activation='relu'),
    keras.layers.Dense(10, activation='softmax')
])

# 5. Compile the model

model.compile(
    optimizer='sgd',
    loss='sparse_categorical_crossentropy',
    metrics=['accuracy']
)

# 6. Train the model

history = model.fit(
    x_train,
    y_train,
    epochs=10
)

# 7. Evaluate the model

test_loss, test_acc = model.evaluate(x_test, y_test)

print("Loss = %.3f" % test_loss)
print("Accuracy = %.3f" % test_acc)

# 8. Display a random test image

n = random.randint(0, 9999)

plt.imshow(x_test[n], cmap='gray')
plt.show()

# 9. Define class labels

class_labels = [
    "T-shirt/Top",
    "Trouser",
    "Pullover",
    "Dress",
    "Coat",
    "Sandal",
    "Shirt",
    "Sneaker",
    "Bag",
    "Ankle Boot"
]

# 10. Predict the image

predicted_value = model.predict(x_test)

print("The image is of type = %d" % np.argmax(predicted_value[n]))

print("Predicted Class:", class_labels[np.argmax(predicted_value[n])])

# 11. Plot Model Loss

plt.plot(history.history['loss'])
plt.title("Model Loss")
plt.xlabel("Epoch")
plt.ylabel("Loss")
plt.show()

# 12. Plot Model Accuracy

plt.plot(history.history['accuracy'])
plt.title("Model Accuracy Graph")
plt.ylabel("Accuracy")
plt.xlabel("Epoch")
plt.show()
